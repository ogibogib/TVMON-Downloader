cmake_minimum_required(VERSION 3.22.1)
project(tvmon_http LANGUAGES CXX)

set(CURL_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../../../build/native-deps/2.2.2/${ANDROID_ABI}")
if(NOT EXISTS "${CURL_ROOT}/libcurl-impersonate.a")
    message(FATAL_ERROR "Run android/native-deps.ps1 before building the Android app.")
endif()
add_library(curl_impersonate STATIC IMPORTED)
set_target_properties(curl_impersonate PROPERTIES IMPORTED_LOCATION "${CURL_ROOT}/libcurl-impersonate.a")
add_library(tvmon_http SHARED native_http.cpp)
target_compile_features(tvmon_http PRIVATE cxx_std_17)
target_compile_options(tvmon_http PRIVATE -Wall -Wextra -Werror -fvisibility=hidden)
target_include_directories(tvmon_http PRIVATE "${CURL_ROOT}/include")
target_link_options(tvmon_http PRIVATE "-Wl,-z,max-page-size=16384" "-Wl,-z,common-page-size=16384" "-Wl,--exclude-libs,ALL")
target_link_libraries(tvmon_http PRIVATE curl_impersonate)
